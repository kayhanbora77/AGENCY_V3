--RiyaIndia Cancel,Diversion,Delay List
WITH Src AS (
    SELECT *
    FROM TA_STANDARD_RIYAINDIA_VF_RESULT AS vf_result
    WHERE vf_result.EUEligible is TRUE AND
    	vf_result.Status IN ('cancel','diversion','Delay') AND
      NOT EXISTS (
            SELECT 1
            FROM RIYAINDIA_MISSCONNECTION_CHECKEDV1 AS mc_checked
            WHERE mc_checked.Id = vf_result.Id
      )
)
--SELECT ConnectionId,FlightNumber,AirlineCode,FromAirport,ToAirport,LegNo,Status,IsMissConnection,EUEligible
SELECT *
FROM TA_STANDARD_RIYAINDIA_VF_RESULT
WHERE ConnectionID IN (
    SELECT DISTINCT ConnectionID
    FROM Src
)
ORDER BY ConnectionID, LegNo;
