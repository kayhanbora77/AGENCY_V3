COPY (
    SELECT
        Id,
        TRIM(REPLACE(BookingRef, ',', ''))                      AS BookingRef,
        CAST(FlightNumber AS VARCHAR)                           AS FlightNumber,
        DepartureDate,
        REGEXP_REPLACE(CAST(ETicketNumber AS VARCHAR), '\.0$', '') AS ETicketNumber
    FROM THOMASCOOK_CASES
)
TO 'C:\Users\cagri\Desktop\tmp\THOMASCOOK_CASES.csv'
(HEADER, DELIMITER ',');
